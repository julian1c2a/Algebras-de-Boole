\section{Introducción}
El álgebra de Boole, originalmente introducida por George Boole en su obra de 1854 \textit{The Laws of Thought}, fue posteriormente axiomatizada de forma rigurosa. En 1904, el matemático estadounidense Edward Vermilye Huntington presentó un conjunto de postulados independientes que definen formalmente una estructura de álgebra de Boole. 

Para evitar cualquier confusión conceptual con la aritmética tradicional, en esta fase inicial emplearemos la signatura propia de la teoría de retículos, utilizando los símbolos $\vee$ (supremo o join) y $\wedge$ (ínfimo o meet), junto a los elementos constantes $\bot$ (mínimo) y $\top$ (máximo). Más adelante, y por conveniencia práctica, transitaremos hacia la notación clásica de sistemas digitales ($+$, $\cdot$, $0$, $1$).

\section{Pre-Axiomas de la Estructura}
Antes de enunciar los postulados, debemos definir rigurosamente sobre qué elementos y operaciones estamos trabajando. 

Partimos de un ente matemático $\mathbb{B}$.
\begin{itemize}
    \item \textbf{[EsConjunto($\mathbb{B}$)]}: Se requiere que $\mathbb{B}$ sea un conjunto.
    \item \textbf{[Constantes]}: Este conjunto ha de cumplir que tiene dos elementos que llamaremos constantes, tales que $\bot \in \mathbb{B}$ y $\top \in \mathbb{B}$. En principio, no asumimos nada sobre la igualdad o desigualdad de estas constantes.
\end{itemize}

Además, vamos a definir dos operaciones binarias internas que denotaremos por $\vee$ y $\wedge$. Estas deben satisfacer rigurosamente la definición de función:
\begin{itemize}
    \item \textbf{[OpBinInt\_$\vee$]}: $\vee : \mathbb{B} \times \mathbb{B} \to \mathbb{B}$ es una operación binaria interna.
    \item \textbf{[OpBinInt\_$\wedge$]}: $\wedge : \mathbb{B} \times \mathbb{B} \to \mathbb{B}$ es una operación binaria interna.
\end{itemize}

Para poder usar estos conceptos con mayor seguridad y flexibilidad en las futuras demostraciones formales, asignaremos nombres cortos a las condiciones de existencia y unicidad de la imagen para estas operaciones:

\begin{itemize}
    \item \textbf{[Existencia\_$\vee$]}: Para todo par existe imagen en $\mathbb{B}$. Es decir, $\forall \langle a,b \rangle \in \mathbb{B} \times \mathbb{B}$, $\exists c \in \mathbb{B}$ tal que $a \vee b = c$.
    \item \textbf{[Existencia\_$\wedge$]}: Análogamente, $\forall \langle a,b \rangle \in \mathbb{B} \times \mathbb{B}$, $\exists d \in \mathbb{B}$ tal que $a \wedge b = d$.
    \item \textbf{[Unicidad\_$\vee$]}: Para un par solo existe una imagen. Esto es, si $a \vee b = c$ y $a \vee b = d$, entonces $c = d$.
    \item \textbf{[Unicidad\_$\wedge$]}: De igual forma para el ínfimo, si $a \wedge b = c$ y $a \wedge b = d$, entonces $c = d$.
\end{itemize}

\section{Los Postulados de Huntington (1904)}
Sobre el sistema $(\mathbb{B}, \vee, \wedge, \bot, \top)$ que cumple los pre-axiomas anteriores, diremos que forma un álgebra de Boole si satisface los siguientes postulados:

\begin{postulado}[Elemento neutro $\vee$ - $ElemNeu_\vee$] \label{post:neutro_vee}
Para todo elemento del conjunto se tiene que:
\[ \forall a \in \mathbb{B}, \quad a \vee \bot = a \]
\end{postulado}

\begin{postulado}[Elemento neutro $\wedge$ - $ElemNeu_\wedge$] \label{post:neutro_wedge}
Para todo elemento del conjunto se tiene que:
\[ \forall a \in \mathbb{B}, \quad a \wedge \top = a \]
\end{postulado}

\begin{postulado}[Conmutatividad $\vee$ - $Comm_\vee$] \label{post:conmut_vee}
Para cualquier par de elementos del conjunto se cumple:
\[ \forall a, b \in \mathbb{B}, \quad a \vee b = b \vee a \]
\end{postulado}

\begin{postulado}[Conmutatividad $\wedge$ - $Comm_\wedge$] \label{post:conmut_wedge}
Para cualquier par de elementos del conjunto se cumple:
\[ \forall a, b \in \mathbb{B}, \quad a \wedge b = b \wedge a \]
\end{postulado}

\begin{postulado}[Distributividad $\vee$ sobre $\wedge$ - $Dist_\vee$] \label{post:distrib_vee_wedge}
Para cualesquiera tres elementos del conjunto se verifica:
\[ \forall a, b, c \in \mathbb{B}, \quad a \vee (b \wedge c) = (a \vee b) \wedge (a \vee c) \]
\end{postulado}

\begin{postulado}[Distributividad $\wedge$ sobre $\vee$ - $Dist_\wedge$] \label{post:distrib_wedge_vee}
Para cualesquiera tres elementos del conjunto se verifica:
\[ \forall a, b, c \in \mathbb{B}, \quad a \wedge (b \vee c) = (a \wedge b) \vee (a \wedge c) \]
\end{postulado}

\begin{postulado}[Complementario - $Comp_\vee, Comp_\wedge$] \label{post:comp}
Para cada elemento del conjunto, existe al menos un elemento complementario tal que:
\begin{align*}
\forall a \in \mathbb{B}, \exists b \in \mathbb{B} \quad : \quad a \vee b &= \top \quad (Comp_\vee) \\
a \wedge b &= \bot \quad (Comp_\wedge)
\end{align*}
\end{postulado}

\textit{Nota: A diferencia de algunas formulaciones clásicas que imponen un axioma de cardinalidad ($\bot \neq \top$) para evitar el álgebra trivial, en este desarrollo permitiremos la existencia del álgebra trivial.}

\section{Teoremas Principales Derivados}

\begin{teorema}[Unicidad de los elementos neutros - $Unic_e, Unic_u$]
\begin{align*}
    \exists! e \in \mathbb{B}, \forall a \in \mathbb{B}, a \vee e &= a \implies e = \bot \quad (Unic_e) \\
    \exists! u \in \mathbb{B}, \forall a \in \mathbb{B}, a \wedge u &= a \implies u = \top \quad (Unic_u)
\end{align*}
\end{teorema}
\begin{proof}[Demostración de $Unic_e$]
Sea $e \in \mathbb{B}$ tal que $\forall a \in \mathbb{B}, a \vee e = a$. Tomando $a = \bot$:
\begin{align*}
    e &= e \vee \bot & (ElemNeu_\vee) \\
      &= \bot \vee e & (Comm_\vee) \\
      &= \bot & (\text{Hipótesis sobre } e)
\end{align*}
La demostración para $Unic_u$ es análoga por dualidad.
\end{proof}

\begin{teorema}[Idempotencia - $Idemp_\vee, Idemp_\wedge$]
\begin{align*}
    \forall a \in \mathbb{B}, \quad a \vee a &= a \quad (Idemp_\vee) \\
    \forall a \in \mathbb{B}, \quad a \wedge a &= a \quad (Idemp_\wedge)
\end{align*}
\end{teorema}
\begin{proof}[Demostración de $Idemp_\vee$]
\begin{align*}
    a &= a \vee \bot & (ElemNeu_\vee) \\
      &= a \vee (a \wedge a') & (Comp_\wedge) \\
      &= (a \vee a) \wedge (a \vee a') & (Dist_\vee) \\
      &= (a \vee a) \wedge \top & (Comp_\vee) \\
      &= \top \wedge (a \vee a) & (Comm_\wedge) \\
      &= a \vee a & (ElemNeu_\wedge)
\end{align*}
La demostración para $Idemp_\wedge$ es análoga.
\end{proof}

\begin{teorema}[Elementos absorbentes - $Abs_{\bot}, Abs_{\top}$]
\begin{align*}
    \forall a \in \mathbb{B}, \quad a \vee \top &= \top \\
    \forall a \in \mathbb{B}, \quad a \wedge \bot &= \bot
\end{align*}
\end{teorema}
\begin{proof}[Demostración para $\top$]
\begin{align*}
    a \vee \top &= (a \vee \top) \wedge \top & (ElemNeu_\wedge) \\
               &= (a \vee \top) \wedge (a \vee a') & (Comp_\vee) \\
               &= a \vee (\top \wedge a') & (Dist_\vee) \\
               &= a \vee (a' \wedge \top) & (Comm_\wedge) \\
               &= a \vee a' & (ElemNeu_\wedge) \\
               &= \top & (Comp_\vee)
\end{align*}
Análogamente para $a \wedge \bot = \bot$.
\end{proof}

\begin{teorema}[Propiedades de absorción - $Abs_{\vee}, Abs_{\wedge}$]
\begin{align*}
    \forall a, b \in \mathbb{B}, \quad a \vee (a \wedge b) &= a \\
    \forall a, b \in \mathbb{B}, \quad a \wedge (a \vee b) &= a
\end{align*}
\end{teorema}
\begin{proof}[Demostración de $Abs_\vee$]
\begin{align*}
    a \vee (a \wedge b) &= (a \wedge \top) \vee (a \wedge b) & (ElemNeu_\wedge) \\
                        &= a \wedge (\top \vee b) & (Dist_\wedge) \\
                        &= a \wedge (b \vee \top) & (Comm_\vee) \\
                        &= a \wedge \top & (Abs_\top) \\
                        &= a & (ElemNeu_\wedge)
\end{align*}
Análogamente se demuestra $Abs_\wedge$.
\end{proof}

\begin{teorema}[Propiedades de orden de retículo - $Prop_{\vee,\wedge}$]
Para cualesquiera $a, b \in \mathbb{B}$:
$$ a \vee b = a \iff a \wedge b = b $$
\end{teorema}
\begin{proof}
Supongamos que $a \vee b = a$.
\begin{align*}
    a \wedge b &= b \wedge a & (Comm_\wedge) \\
               &= b \wedge (a \vee b) & (\text{Hipótesis } a \vee b = a) \\
               &= b & (Abs_\wedge)
\end{align*}
La implicación recíproca se demuestra de forma análoga asumiendo $a \wedge b = b$.
\end{proof}

\begin{teorema}[Equivalencia de operaciones - $Equa_{\vee,\wedge}$]
Para cualesquiera $a, b \in \mathbb{B}$:
$$ a \vee b = a \wedge b \implies a = b $$
\end{teorema}
\begin{proof}
Supongamos $a \vee b = a \wedge b$. Observamos que:
\begin{align*}
    a &= a \vee (a \wedge b) & (Abs_\vee) \\
      &= a \vee (a \vee b) & (\text{Hipótesis})
\end{align*}
Aplicando $Prop_{\vee,\wedge}$, dado que $a \vee (a \vee b) = a$, deducimos que $a \wedge (a \vee b) = a \vee b$.
\begin{align*}
    a \vee b &= a \wedge (a \vee b) & (\text{Resultado anterior}) \\
             &= a & (Abs_\wedge)
\end{align*}
De manera simétrica para $b$:
\begin{align*}
    b &= b \vee (a \wedge b) & (Abs_\vee) \\
      &= b \vee (a \vee b) & (\text{Hipótesis}) \\
      &= (a \vee b) \vee b & (Comm_\vee)
\end{align*}
Aplicando de nuevo $Prop_{\vee,\wedge}$ sobre esta igualdad, obtenemos que $(a \vee b) \wedge b = a \vee b$. Pero sabemos por $Abs_\wedge$ que $(a \vee b) \wedge b = b \wedge (a \vee b) = b$. 
Por consiguiente, $a \vee b = b$. Finalmente, uniendo ambos resultados: $a = a \vee b = b$.
\end{proof}

\begin{teorema}[Teorema de Cancelación - $Equa_{canc}$]
Para cualesquiera $a, b, c \in \mathbb{B}$:
$$ a \vee b = a \vee c \quad \text{y} \quad a \wedge b = a \wedge c \implies b = c $$
\end{teorema}
\begin{proof}
\begin{align*}
    b &= b \wedge (a \vee b) & (Abs_\wedge) \\
      &= b \wedge (a \vee c) & (\text{Hipótesis } a \vee b = a \vee c) \\
      &= (b \wedge a) \vee (b \wedge c) & (Dist_\wedge) \\
      &= (a \wedge b) \vee (b \wedge c) & (Comm_\wedge) \\
      &= (a \wedge c) \vee (b \wedge c) & (\text{Hipótesis } a \wedge b = a \wedge c) \\
      &= (c \wedge a) \vee (c \wedge b) & (Comm_\wedge \text{ aplicado dos veces}) \\
      &= c \wedge (a \vee b) & (Dist_\wedge) \\
      &= c \wedge (a \vee c) & (\text{Hipótesis } a \vee b = a \vee c) \\
      &= c & (Abs_\wedge)
\end{align*}
\end{proof}

\begin{teorema}[Unicidad del complemento - $Unic_{comp}$]
Para todo $a \in \mathbb{B}$, su complemento $a'$ es único. Si existe $x \in \mathbb{B}$ tal que $a \vee x = \top$ y $a \wedge x = \bot$, entonces $x = a'$.
\end{teorema}
\begin{proof}
\begin{align*}
    x &= x \wedge \top & (ElemNeu_\wedge) \\
      &= x \wedge (a \vee a') & (Comp_\vee) \\
      &= (x \wedge a) \vee (x \wedge a') & (Dist_\wedge) \\
      &= (a \wedge x) \vee (x \wedge a') & (Comm_\wedge) \\
      &= \bot \vee (x \wedge a') & (\text{Hipótesis } a \wedge x = \bot) \\
      &= (a \wedge a') \vee (x \wedge a') & (Comp_\wedge) \\
      &= (a \vee x) \wedge a' & (Dist_\wedge) \\
      &= \top \wedge a' & (\text{Hipótesis } a \vee x = \top) \\
      &= a' \wedge \top & (Comm_\wedge) \\
      &= a' & (ElemNeu_\wedge)
\end{align*}
\end{proof}

\begin{teorema}[Involución - $Comp_{inv}$]
Para todo $a \in \mathbb{B}$, $(a')' = a$.
\end{teorema}
\begin{proof}
Por definición, el complemento de $a'$, denotado como $(a')'$, es el elemento único que satisface:
$$ a' \vee (a')' = \top \quad \text{y} \quad a' \wedge (a')' = \bot $$
Sin embargo, por la conmutatividad ($Comm_\vee$ y $Comm_\wedge$), sabemos que:
$$ a' \vee a = a \vee a' = \top \quad \text{y} \quad a' \wedge a = a \wedge a' = \bot $$
Esto demuestra que $a$ actúa como un complemento de $a'$. Por el teorema de unicidad del complemento ($Unic_{comp}$), concluimos necesariamente que $(a')' = a$.
\end{proof}

\begin{teorema}[Leyes de De Morgan - $Mor_{\vee, \wedge}$]
Para cualesquiera $a, b \in \mathbb{B}$:
\begin{align*}
    (a \vee b)' &= a' \wedge b' \\
    (a \wedge b)' &= a' \vee b'
\end{align*}
\end{teorema}
\begin{proof}
(Pendiente de expandir en base a la distributividad y propiedades previas, comprobando que $(a \vee b) \vee (a' \wedge b') = \top$ y $(a \vee b) \wedge (a' \wedge b') = \bot$, para luego aplicar $Unic_{comp}$).
\end{proof}

\begin{teorema}[Condición de Álgebra Trivial]
Si los elementos neutros son iguales ($\bot = \top$), entonces el álgebra de Boole tiene un único elemento, es decir, $\mathbb{B} = \{\bot\} = \{\top\}$.
\end{teorema}
\begin{proof}
Supongamos que $\bot = \top$. Sea $x \in \mathbb{B}$ un elemento cualquiera:
\begin{align*}
    x &= x \vee \bot & (ElemNeu_\vee) \\
      &= x \vee \top & (\text{Hipótesis } \bot = \top) \\
      &= \top & (Abs_\top)
\end{align*}
Por tanto, todo elemento $x$ del conjunto es idéntico a $\top$. Esto implica que el único elemento del conjunto es $\top$ (que a su vez es igual a $\bot$). Así, $\mathbb{B} = \{\top\} = \{\bot\}$.
\end{proof}

\begin{teorema}[Complemento Idéntico implica Álgebra Trivial]
Si existe un elemento $a \in \mathbb{B}$ que es igual a su propio complemento (es decir, $a' = a$), entonces $\mathbb{B} = \{\top\} = \{\bot\}$.
\end{teorema}
\begin{proof}
Supongamos que existe $a \in \mathbb{B}$ tal que $a' = a$. Por el postulado del Complemento ($Comp_\vee$ y $Comp_\wedge$), sabemos que:
$$ a \vee a' = \top \quad \text{y} \quad a \wedge a' = \bot $$
Sustituyendo la hipótesis $a' = a$ en ambas ecuaciones, obtenemos:
$$ a \vee a = \top \quad \text{y} \quad a \wedge a = \bot $$
Aplicando el teorema de Idempotencia ($Idemp_\vee$ y $Idemp_\wedge$), sabemos que $a \vee a = a$ y $a \wedge a = a$. Por tanto:
$$ a = \top \quad \text{y} \quad a = \bot $$
Lo cual implica que $\bot = \top$. Aplicando el teorema anterior (Condición de Álgebra Trivial), concluimos que $\mathbb{B} = \{\top\} = \{\bot\}$.
\end{proof}
